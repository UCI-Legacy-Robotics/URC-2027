cmake_minimum_required(VERSION 3.8)
project(arm_gamepad_teleop)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
add_executable(arm_teleop src/arm_teleop.cpp)
target_compile_features(arm_teleop PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(arm_teleop PRIVATE -Wall -Wextra -Wpedantic)
endif()
ament_target_dependencies(arm_teleop rclcpp sensor_msgs control_msgs geometry_msgs moveit_msgs)
install(TARGETS arm_teleop DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
